// PROJECT DEADZONE cinematic soundtrack test v0.2
// Ambience Mini 2.7.x

playlist menu = [
  "menu_dead_signal.ogg" <-4.0>
];

playlist settlement = [
  "settlement_last_light.ogg" <-4.0>
];

// Location themes deliberately reuse the accepted cinematic set for this
// first integration pass. They can be replaced one by one without changing
// the server-side detection logic.
playlist camp_theme = [ "theme_camp_last_light.ogg" <-4.5>, "theme_camp_dawn_patrol.ogg" <-4.5> ];
playlist survivor_theme = [ "theme_survivor_open_road.ogg" <-5.0>, "theme_survivor_road_embers.ogg" <-5.0> ];
playlist cdf_theme = [ "theme_cdf_watchline.ogg" <-5.0>, "theme_cdf_last_watch.ogg" <-5.0> ];
playlist raider_theme = [ "theme_raider_rust_and_dust.ogg" <-4.5>, "theme_raider_scrap_horizon.ogg" <-4.5> ];
playlist remnant_theme = [ "theme_remnant_iron_memory.ogg" <-5.0>, "theme_remnant_faded_orders.ogg" <-5.0> ];
playlist aegis_theme = [ "theme_aegis_cold_glass.ogg" <-5.0>, "theme_aegis_white_room.ogg" <-5.0> ];
playlist warden_theme = [ "theme_warden_machine_vigil.ogg" <-5.0>, "theme_warden_ghost_signal.ogg" <-5.0> ];
playlist infected_theme = [ "theme_infected_hollow_breath.ogg" <-5.0>, "theme_infected_quiet_hunger.ogg" <-5.0> ];

// Named encounters are short high-priority interruptions. When they end,
// Ambience Mini resumes the paused location theme instead of starting over.
playlist named_survivor = [ "combat_better_future.ogg" <-3.5> ];
playlist named_cdf = [ "named_cdf_brass_vanguard.ogg" <-3.5> ];
playlist named_raider = [ "named_raider_iron_jackal.ogg" <-3.0> ];
playlist named_remnant = [ "named_remnant_red_echo.ogg" <-3.0> ];
playlist named_aegis = [ "named_aegis_glass_dominion.ogg" <-2.5> ];
playlist named_warden = [ "named_warden_signal_duel.ogg" <-2.5> ];
playlist named_infected = [ "combat_infected_swarm_alarm.ogg" <-3.0>, "combat_infected_close_breath.ogg" <-3.0> ];
playlist named_unknown = [ "combat_contact.ogg" <-3.0>, "combat_orchestral_assault.ogg" <-3.5> ];

playlist boss_01 = [ "boss_01_axel_fuel_king.ogg" <-2.0> ];
playlist boss_02 = [ "boss_02_argus_fragment_broken_directive.ogg" <-2.5> ];
playlist boss_03 = [ "boss_03_choir_vessel_black_resonance.ogg" <-2.0> ];
playlist boss_04 = [ "boss_04_ash_captain_cinderline.ogg" <-2.0> ];
playlist boss_05 = [ "boss_05_fuel_route_scout_dust_chase.ogg" <-2.0> ];
playlist boss_06 = [ "boss_06_brass_hound_overheat.ogg" <-2.0> ];
playlist boss_07 = [ "boss_07_white_stitch_false_triage.ogg" <-2.5> ];
playlist boss_08 = [ "boss_08_marshal_graves_lockdown.ogg" <-2.0> ];
playlist boss_09 = [ "boss_09_primordial_first_hunger.ogg" <-2.0> ];
playlist boss_10 = [ "boss_10_echo7_dead_channel.ogg" <-2.5> ];
playlist boss_11 = [ "boss_11_reactor_saint_critical_mass.ogg" <-2.0> ];
playlist boss_12 = [ "boss_12_siege01_break_the_gate.ogg" <-2.0> ];
playlist boss_13 = [ "boss_13_ancient_abomination_spore_crown.ogg" <-2.0> ];

playlist city = [
  "city_empty_windows.ogg" <-5.0>,
  "city_broken_signal.ogg" <-5.0>
];

playlist wild = [
  "wild_long_road.ogg" <-5.0>
];

playlist rain = [
  "wild_after_rain.ogg" <-5.0>
];

playlist desert = [
  "desert_dustline.ogg" <-5.0>
];

playlist ocean = [
  "ocean_cold_horizon.ogg" <-5.0>
];

playlist underground = [
  "underground_below_grid.ogg" <-4.0>
];

playlist night = [
  "night_watch.ogg" <-5.0>
];

// Natural exploration states share one playlist. Rain/night/biome/cave
// boundaries therefore no longer restart the music every few seconds; the
// next track still rotates through the full environmental catalogue.
playlist exploration = city ++ wild ++ rain ++ desert ++ ocean ++ underground ++ night;

playlist combat = [
  "combat_contact.ogg" <-3.0>,
  "combat_heavy_battle.ogg" <-3.0>,
  "combat_better_future.ogg" <-3.0>,
  "combat_orchestral_assault.ogg" <-3.5>,
  "combat_infected_swarm_alarm.ogg" <-3.0>,
  "combat_infected_close_breath.ogg" <-3.0>
];

playlist boss = [
  "boss_red_protocol.ogg" <-2.0>,
  "boss_determined_pursuit.ogg" <-2.5>,
  "boss_last_stand.ogg" <-2.0>
];

playlist daylight = city ++ wild;

begin
  when ( @main_menu ) play menu;
  when ( @joining ) play menu;
  when ( @credits ) play [ ];

  when ( @in_game )
  begin
    when ( @dead ) play instant [ ];

    // PDZ bosses have individual cues. Generic mobs with a boss-like health
    // bar never enter this layer unless the server identifies the encounter.
    when ( any effect in $effects has effect == "project_deadzone:music_boss_01" end ) play boss_01 priority 8;
    when ( any effect in $effects has effect == "project_deadzone:music_boss_02" end ) play boss_02 priority 8;
    when ( any effect in $effects has effect == "project_deadzone:music_boss_03" end ) play boss_03 priority 8;
    when ( any effect in $effects has effect == "project_deadzone:music_boss_04" end ) play boss_04 priority 8;
    when ( any effect in $effects has effect == "project_deadzone:music_boss_05" end ) play boss_05 priority 8;
    when ( any effect in $effects has effect == "project_deadzone:music_boss_06" end ) play boss_06 priority 8;
    when ( any effect in $effects has effect == "project_deadzone:music_boss_07" end ) play boss_07 priority 8;
    when ( any effect in $effects has effect == "project_deadzone:music_boss_08" end ) play boss_08 priority 8;
    when ( any effect in $effects has effect == "project_deadzone:music_boss_09" end ) play boss_09 priority 8;
    when ( any effect in $effects has effect == "project_deadzone:music_boss_10" end ) play boss_10 priority 8;
    when ( any effect in $effects has effect == "project_deadzone:music_boss_11" end ) play boss_11 priority 8;
    when ( any effect in $effects has effect == "project_deadzone:music_boss_12" end ) play boss_12 priority 8;
    when ( any effect in $effects has effect == "project_deadzone:music_boss_13" end ) play boss_13 priority 8;

    // Named threats override everything, but do not hard-cut. A lower
    // priority location track is paused and resumes after the threat leaves.
    when ( any effect in $effects has effect == "project_deadzone:music_named_survivor" end ) play named_survivor priority 6;
    when ( any effect in $effects has effect == "project_deadzone:music_named_cdf" end ) play named_cdf priority 6;
    when ( any effect in $effects has effect == "project_deadzone:music_named_raider" end ) play named_raider priority 6;
    when ( any effect in $effects has effect == "project_deadzone:music_named_remnant" end ) play named_remnant priority 6;
    when ( any effect in $effects has effect == "project_deadzone:music_named_aegis" end ) play named_aegis priority 6;
    when ( any effect in $effects has effect == "project_deadzone:music_named_warden" end ) play named_warden priority 6;
    when ( any effect in $effects has effect == "project_deadzone:music_named_infected" end ) play named_infected priority 6;
    when ( any effect in $effects has effect == "project_deadzone:music_named_unknown" end ) play named_unknown priority 6;

    // Generic threat music wins over location music.
    when ( @boss_fight ) play combat priority 4;
    when ( @in_combat ) play combat priority 4;

    // Per-player location states supplied by the PDZ server bridge.
    when ( any effect in $effects has effect == "project_deadzone:music_camp" end ) play camp_theme priority 1;
    when ( any effect in $effects has effect == "project_deadzone:music_survivor" end ) play survivor_theme priority 1;
    when ( any effect in $effects has effect == "project_deadzone:music_cdf" end ) play cdf_theme priority 1;
    when ( any effect in $effects has effect == "project_deadzone:music_raider" end ) play raider_theme priority 1;
    when ( any effect in $effects has effect == "project_deadzone:music_remnant" end ) play remnant_theme priority 1;
    when ( any effect in $effects has effect == "project_deadzone:music_aegis" end ) play aegis_theme priority 1;
    when ( any effect in $effects has effect == "project_deadzone:music_warden" end ) play warden_theme priority 1;
    when ( any effect in $effects has effect == "project_deadzone:music_infected" end ) play infected_theme priority 1;

    // All natural exploration contexts use one stable rotation. Dedicated
    // camp/site effects above and combat priorities remain contextual.
    play exploration;
  end
end
